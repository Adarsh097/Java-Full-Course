import java.util.*;

class Document {

    private int id;
    private String title;
    private String folderName;
    private int pages;

    // Constructor
    public Document(int id, String title, String folderName, int pages) {
        this.id = id;
        this.title = title;
        this.folderName = folderName;
        this.pages = pages;
    }

    // Getters
    public int getId() {
        return id;
    }

    public String getTitle() {
        return title;
    }

    public String getFolderName() {
        return folderName;
    }

    public int getPages() {
        return pages;
    }

    // Setters
    public void setId(int id) {
        this.id = id;
    }

    public void setTitle(String title) {
        this.title = title;
    }

    public void setFolderName(String folderName) {
        this.folderName = folderName;
    }

    public void setPages(int pages) {
        this.pages = pages;
    }
}

public class Solution {

    public static Document[] docsWithOddPages(Document[] docs) {

        ArrayList<Document> list = new ArrayList<>();

        for (Document d : docs) {
            if (d.getPages() % 2 != 0) {
                list.add(d);
            }
        }

        list.sort((a, b) -> a.getId() - b.getId());

        Document[] ans = new Document[list.size()];

        for (int i = 0; i < list.size(); i++) {
            ans[i] = list.get(i);
        }

        return ans;
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        Document[] docs = new Document[4];

        for (int i = 0; i < 4; i++) {

            int id = Integer.parseInt(sc.nextLine());
            String title = sc.nextLine();
            String folder = sc.nextLine();
            int pages = Integer.parseInt(sc.nextLine());

            docs[i] = new Document(id, title, folder, pages);
        }

        Document[] result = docsWithOddPages(docs);

        for (Document d : result) {
            System.out.println(d.getId() + " "
                    + d.getTitle() + " "
                    + d.getFolderName() + " "
                    + d.getPages());
        }
    }
}
